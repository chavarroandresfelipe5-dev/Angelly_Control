<script>

const SERVICES=[

{"code":1,"name":"Manicure","price":27000},

{"code":2,"name":"Manicure secado rápido","price":32000},

{"code":3,"name":"Base Rubber","price":83000},

{"code":4,"name":"Press On","price":128000},

{"code":5,"name":"Acrílico o Polygel","price":165000},

{"code":6,"name":"Dipping","price":95000},

{"code":7,"name":"Baño de acrílico","price":85000},

{"code":8,"name":"Manicure niñas","price":18000},

{"code":9,"name":"Semipermanente","price":65000},

{"code":10,"name":"Retiro acrílico o gel","price":37000},

{"code":11,"name":"Retiro semipermanente","price":15000},

{"code":12,"name":"Spa de pies","price":50000},

{"code":13,"name":"Pedicure","price":35000},

{"code":14,"name":"Pedicure secado rapido","price":40000},

{"code":15,"name":"Pedicure semipermantente","price":67000},

{"code":16,"name":"Shampoo","price":15000},

{"code":17,"name":"Arreglos","price":5000},

{"code":0,"name":"NADA ESE DIA","price":0}

];


let records=
JSON.parse(
localStorage.getItem("angelly_records_v2")||"[]"
);


const money=n=>
new Intl.NumberFormat(
"es-CO",
{
style:"currency",
currency:"COP",
maximumFractionDigits:0
}
).format(Math.round(n||0));


const $=id=>
document.getElementById(id);


function findService(){

const code=
Number($("code").value);

return SERVICES.find(
s=>s.code===code
);

}


function calc(){

const s=findService();

const q=
Math.max(
1,
Number($("qty").value)||1
);


if(!s){

$("serviceName").textContent=
$("code").value
?"Código no encontrado"
:"Escribe un código";

$("servicePrice").textContent="";

$("status").textContent="";


["gross","discount","net","mine","other"]
.forEach(
x=>$(x).textContent=money(0)
);

return;

}


$("serviceName").textContent=s.name;

$("servicePrice").textContent=
money(s.price);

$("status").textContent=
"✓ Servicio encontrado automáticamente";

$("status").className="note ok";


const gross=
s.price*q;

const discount=
gross*.04;

const net=
gross-discount;


$("gross").textContent=
money(gross);

$("discount").textContent=
money(discount);

$("net").textContent=
money(net);

$("mine").textContent=
money(net*.60);

$("other").textContent=
money(net*.40);

}


$("code")
.addEventListener("input",calc);


$("qty")
.addEventListener("input",calc);


$("save")
.addEventListener("click",()=>{

const s=findService();

const q=
Math.max(
1,
Number($("qty").value)||1
);


if(!s || s.code===0){

alert(
"Escribe un código de servicio válido."
);

return;

}


const gross=
s.price*q;

const discount=
gross*.04;

const net=
gross-discount;


records.unshift({

date:new Date().toISOString(),

code:s.code,

service:s.name,

client:$("client").value.trim(),

qty:q,

gross:gross,

discount:discount,

net:net,

mine:net*.6,

other:net*.4

});


localStorage.setItem(
"angelly_records_v2",
JSON.stringify(records)
);


$("client").value="";

$("code").value="";

$("qty").value=1;

calc();

render();


});


function sameDay(a,b){

return

a.getFullYear()===b.getFullYear()

&&

a.getMonth()===b.getMonth()

&&

a.getDate()===b.getDate();

}


function render(){

const now=new Date();


const today=
records.filter(
r=>sameDay(
new Date(r.date),
now
)
);


const month=
records.filter(r=>{

let d=new Date(r.date);

return

d.getFullYear()===now.getFullYear()

&&

d.getMonth()===now.getMonth();

});


$("today").textContent=
money(
today.reduce(
(a,r)=>a+r.net,
0
)
);


$("todayMine").textContent=
money(
today.reduce(
(a,r)=>a+r.mine,
0
)
);


$("count").textContent=
today.length;


$("monthMine").textContent=
money(
month.reduce(
(a,r)=>a+r.mine,
0
)
);


$("history").innerHTML=

records
.slice(0,30)
.map(r=>`

<tr>

<td>
${new Date(r.date)
.toLocaleTimeString(
"es-CO",
{
hour:"2-digit",
minute:"2-digit"
}
)}
</td>

<td>${r.code}</td>

<td>${r.service}</td>

<td>${money(r.net)}</td>

<td>${money(r.mine)}</td>

</tr>

`)
.join("")

||

"<tr><td colspan='5'>Todavía no hay registros.</td></tr>";

}


$("export")
.addEventListener("click",()=>{

const rows=[

[
"Fecha",
"Código",
"Servicio",
"Cliente",
"Cantidad",
"Valor servicio",
"Descuento 4%",
"Después 4%",
"Tu 60%",
"Otro 40%"
],

...

records.map(r=>[

new Date(r.date)
.toLocaleString("es-CO"),

r.code,

r.service,

r.client,

r.qty,

r.gross,

r.discount,

r.net,

r.mine,

r.other

])

];


const csv=
"\ufeff"+
rows
.map(
row=>
row
.map(
v=>
`"${String(v??"")
.replaceAll('"','""')}"`
)
.join(";")
)
.join("\n");


const blob=
new Blob(
[csv],
{
type:"text/csv;charset=utf-8"
}
);


const a=
document.createElement("a");

a.href=
URL.createObjectURL(blob);

a.download=
"CONTROL_ANGELLY.csv";

a.click();

URL.revokeObjectURL(a.href);

});


if("serviceWorker" in navigator){

navigator.serviceWorker.register("sw.js");

}


render();

calc();

</script>
